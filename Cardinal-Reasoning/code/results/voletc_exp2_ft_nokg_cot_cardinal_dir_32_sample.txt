def find_nearest_place(point, places):
    min_dist = None
    nearest = None
    for place, coords in places.items():
        dist = haversine(point, coords)
        if min_dist is None or dist < min_dist:

            nearest = place
    return nearest, min_dist

def compute_bearing(point_a, point_b):
    """
    Compute bearing from point_a to point_b in degrees from North.
    """
    lat1, lon1 = point_a
    lat2, lon2 = point_b
    dlon = math.radians(lon2 - lon1)
    lat1_rad = math.radians(lat1)
    lat2_rad = math.radians(lat2)
    x = math.sin(dlon) * math.cos(lat2_rad)
    y = math.cos(lat1_rad) * math.sin(lat2_rad) - \
        math.sin(lat1_rad) * math.cos(lat2_rad) * math.cos(dlon)
    bearing = math.atan2(x, y)
    bearing = math.degrees(bearing)
    bearing = (bearing + 360) % 360
    return bearing

def offset_point(point, bearing, distance_km):
    """
    Return point offset by distance_km in bearing direction.
    """
    lat, lon = point
    lat_rad = math.radians(lat)
    lon_rad = math.radians(lon)
    bearing_rad = math.radians(bearing)
    d = distance_km / EARTH_RADIUS_KM
    lat2_rad = math.asin(math.sin(lat_rad) * math.cos(d) +
                         math.cos(lat_rad) * math.sin(d) * math.cos(bearing_rad))
    lon2_rad = lon_rad + math.atan2(math.sin(bearing_rad) * math.sin(d) * math.cos(lat_rad),
                                   math.cos(d) - math.sin(lat_rad) * math.sin(lat2_rad))
    lat2 = math.degrees(lat2_rad)
    lon2 = math.degrees(lon2_rad)
    return (lat2, lon2)

def determine_direction(point_a: str, point_b: str) -> str:
    """
    Determine the cardinal direction from point_a to point_b.
    """
    # Find lat/lon of both points
    lat_a, lon_a = get_point(point_a)
    lat_b, lon_b = get_point(point_b)
    # Compute bearing from A to B
    bearing = haversine_bearing(lat_a, lon_a, lat_b, lon_b)
    # Map bearing to cardinal direction
    return bearing_to_cardinal(bearing)

def get_point(name: str) -> Tuple[float, float]:
    """
    Return the latitude and longitude of a named place.
    """
    places = {
        "India": (20.0, 80.0),
        "Australia": (-20.0, 80.0),
    }
    return places[name]

def haversine_bearing(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    """
    Compute the bearing from point 1 to point 2 in degrees.
    0° is north, 90° is east, 180° is south, 270° is west.
    """
    phi1 = math.radians(lat1)
    phi2 = math.radians(lat2)
    dlam = math.radians(lon2 - lon1)
    x = math.sin(dlam) * math.cos(phi2)
    y = math.cos(phi1) * math.sin(phi2) - math.sin(phi1) * math.cos(phi2) * math.cos(dlam)
    bearing = math.atan2(x, y)
    bearing = (bearing + 2 * math.pi) % (2 * math.pi)
    return math.degrees(bearing)

def bearing_to_cardinal(bearing: float) -> str:
    """
    Convert a bearing in degrees to a cardinal direction.
    """
    if 22.5 <= bearing < 112.5:
        return "east_of"
    if 112.5 <= bearing < 157.5:
        return "southeast_of"
    if 157.5 <= bearing < 202.5:
        return "south_of"
    if 202.5 <= bearing < 247.5:
        return "southwest_of"
    if 247.5 <= bearing < 292.5:
        return "west_of"
    if 292.5 <= bearing < 337.5:
        return "northwest_of"
    return "north_of"
